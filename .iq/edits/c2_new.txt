    else mbufnt_take f (f (map hd buf) (hd ts) z) (map tl buf) (hd ts) (tl ts))"

function (sequential) mres_mprev_next ::

     (let (ys, buf', ts') = mres_mprev_next I (Dynamic xs) (t' # ts)
